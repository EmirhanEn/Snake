cmake_minimum_required(VERSION 3.20)
project(MeyveBahcesi VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(game_rules src/game.cpp)
target_include_directories(game_rules PUBLIC src)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
enable_testing()
add_library(snake_ribbon src/snake_ribbon.cpp)
target_include_directories(snake_ribbon PUBLIC src)
add_executable(snake_ribbon_tests tests/snake_ribbon_tests.cpp)
target_link_libraries(snake_ribbon_tests PRIVATE snake_ribbon)
add_test(NAME continuous_snake_body COMMAND snake_ribbon_tests)
add_library(asset_files src/asset_files.cpp)
target_include_directories(asset_files PUBLIC src)
add_executable(asset_file_tests tests/asset_file_tests.cpp)
target_link_libraries(asset_file_tests PRIVATE asset_files)
add_test(NAME asset_file_paths COMMAND asset_file_tests)
add_executable(game_tests tests/game_tests.cpp)
target_link_libraries(game_tests PRIVATE game_rules)
add_test(NAME snake_rules COMMAND game_tests)
add_library(composer src/composer.cpp)
target_include_directories(composer PUBLIC src)
add_executable(audio_tests tests/audio_tests.cpp)
target_link_libraries(audio_tests PRIVATE composer)
add_test(NAME original_audio COMMAND audio_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  set(RAYLIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/raylib-5.5_win64_msvc16")
  add_executable(MeyveBahcesi src/main.cpp src/audio.cpp src/art.cpp)
  target_include_directories(MeyveBahcesi PRIVATE "${RAYLIB_DIR}/include")
  target_link_libraries(MeyveBahcesi PRIVATE game_rules asset_files snake_ribbon "${RAYLIB_DIR}/lib/raylib.lib" winmm opengl32 gdi32 user32 shell32)
  target_link_options(MeyveBahcesi PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
  set_target_properties(MeyveBahcesi PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/dist")
  add_custom_target(package_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_CURRENT_SOURCE_DIR}/dist/assets"
    COMMENT "Copying PNG, WAV and TTF resources next to the game")
  add_dependencies(MeyveBahcesi package_assets)
  add_executable(asset_export tools/export_assets.cpp)
  target_include_directories(asset_export PRIVATE "${RAYLIB_DIR}/include")
  target_link_libraries(asset_export PRIVATE composer "${RAYLIB_DIR}/lib/raylib.lib" winmm opengl32 gdi32 user32 shell32)
  add_executable(media_pack_tests tests/media_pack_tests.cpp)
  target_include_directories(media_pack_tests PRIVATE "${RAYLIB_DIR}/include")
  target_link_libraries(media_pack_tests PRIVATE asset_files "${RAYLIB_DIR}/lib/raylib.lib" winmm opengl32 gdi32 user32 shell32)
  add_test(NAME external_source_media COMMAND media_pack_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_test(NAME external_distribution_media COMMAND media_pack_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/assets")
  add_executable(snake_render_tests tests/snake_render_tests.cpp src/art.cpp)
  target_include_directories(snake_render_tests PRIVATE "${RAYLIB_DIR}/include")
  target_link_libraries(snake_render_tests PRIVATE asset_files snake_ribbon "${RAYLIB_DIR}/lib/raylib.lib" winmm opengl32 gdi32 user32 shell32)
  add_test(NAME rendered_snake_continuity COMMAND snake_render_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
